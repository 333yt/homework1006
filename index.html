    <script>

        let db, auth;

        let currentRoomId = null;

        let currentUserUid = null;

        let isHost = false;

        let myNickName = '';

        let roomRef = null;

        let timerInterval = null;

 

        // 內建題庫 (示範精選題庫)

        const QUESTION_BANK = [

            { id: "A01", type: "party", cat: "搞笑個人挑戰", title: "模仿 3 種動物，讓朋友猜出來", options: null, answer: "完成挑戰" },

            { id: "A02", type: "party", cat: "搞笑個人挑戰", title: "不說話，用動作表演 3 種職業", options: null, answer: "完成挑戰" },

            { id: "B01", type: "party", cat: "趣味投票", title: "旅行會選哪裡？", options: ["山上", "海邊"], answer: null },

            { id: "B02", type: "party", cat: "趣味投票", title: "想擁有哪種超能力？", options: ["飛行", "瞬間移動"], answer: null },

            { id: "C01", type: "pk", cat: "知識問答", title: "太陽系最大的行星是哪一顆？", options: ["地球", "火星", "木星", "土星"], answer: "木星" },

            { id: "C02", type: "pk", cat: "知識問答", title: "水的化學式是什麼？", options: ["H₂O", "CO₂", "NaCl", "O₂"], answer: "H₂O" },

            { id: "C05", type: "pk", cat: "知識問答", title: "平面三角形的內角和是多少？", options: ["90°", "180°", "360°", "270°"], answer: "180°" },

            { id: "C07", type: "pk", cat: "知識問答", title: "日本首都是哪個城市？", options: ["大阪", "京都", "東京", "橫濱"], answer: "東京" }

        ];

 

        // 頁面切換控制

        function switchView(viewId) {

            document.querySelectorAll('main > div').forEach(el => el.classList.add('hidden'));

            document.getElementById(viewId).classList.remove('hidden');

        }

 

        // 初始化 Firebase

        function initFirebaseAndStart() {

            const rawConfig = document.getElementById('firebase-config-input').value.trim();

            try {

                // 嘗試解析 JSON 或物件字串

                let configClean = rawConfig;

                if (!configClean.startsWith('{')) {

                    configClean = '{' + configClean + '}';

                }

                // 簡易安全性評估並初始化

                const firebaseConfig = eval('(' + configClean + ')');

               

                if (!firebase.apps.length) {

                    firebase.initializeApp(firebaseConfig);

                }

                auth = firebase.auth();

                db = firebase.database();

 

                auth.signInAnonymously().then((cred) => {

                    currentUserUid = cred.user.uid;

                    alert('Firebase 連線成功！');

                    switchView('view-home');

                }).catch((error) => {

                    alert('匿名登入失敗: ' + error.message);

                });

            } catch (e) {

                alert('Firebase 設定格式有誤，請確認複製內容是否正確！');

            }

        }

 

        // 產生 6 碼隨機房號

        function generateRoomCode() {

            const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';

            let code = '';

            for (let i = 0; i < 6; i++) {

                code += chars.charAt(Math.floor(Math.random() * chars.length));

            }

            return code;

        }

 

        // 建立房間

        function createRoom() {

            const name = document.getElementById('host-name').value.trim();

            if (!name) { alert('請輸入主持人暱稱！'); return; }

            myNickName = name;

            isHost = true;

 

            currentRoomId = generateRoomCode();

            const mode = document.getElementById('game-mode').value;

            const rounds = parseInt(document.getElementById('game-rounds').value);

            const difficulty = document.getElementById('game-difficulty').value;

 

            roomRef = db.ref('rooms/' + currentRoomId);

           

            roomRef.set({

                meta: {

                    hostUid: currentUserUid,

                    mode: mode,

                    totalRounds: rounds,

                    difficulty: difficulty,

                    status: 'lobby'

                },

                players: {

                    [currentUserUid]: {

                        name: myNickName,

                        ready: true,

                        score: 0,

                        team: 'host'

                    }

                },

                game: {

                    currentRound: 0,

                    status: 'lobby'

                }

            }).then(() => {

                enterLobby();

            });

        }

 

        // 加入房間

        function joinRoom() {

            const code = document.getElementById('join-room-code').value.trim().toUpperCase();

            const name = document.getElementById('player-name').value.trim();

            if (code.length !== 6) { alert('請輸入正確的 6 碼房號！'); return; }

            if (!name) { alert('請輸入你的暱稱！'); return; }

 

            currentRoomId = code;

            myNickName = name;

            isHost = false;

 

            roomRef = db.ref('rooms/' + currentRoomId);

            roomRef.once('value', (snapshot) => {

                if (!snapshot.exists()) {

                    alert('找不到此房間，請檢查房號是否正確！');

                    return;

                }

                const roomData = snapshot.val();

                if (roomData.meta.status !== 'lobby') {

                    alert('遊戲已經開始或已結束，無法加入！');

                    return;

                }

 

                roomRef.child('players/' + currentUserUid).set({

                    name: myNickName,

                    ready: false,

                    score: 0,

                    team: 'player'

                }).then(() => {

                    enterLobby();

                });

            });

        }

 

        // 進入大廳監聽

        function enterLobby() {

            switchView('view-lobby');

            document.getElementById('lobby-code-display').innerText = currentRoomId;

           

            if (isHost) {

                document.getElementById('host-controls').classList.remove('hidden');

                document.getElementById('btn-ready').classList.add('hidden');

            }

 

            // 即時監聽房間與玩家變化

            roomRef.on('value', (snapshot) => {

                if (!snapshot.exists()) return;

                const data = snapshot.val();

               

                // 更新玩家列表

                const playersObj = data.players || {};

                const playerListEl = document.getElementById('player-list');

                playerListEl.innerHTML = '';

                let count = 0;

 

                Object.keys(playersObj).forEach(uid => {

                    count++;

                    const p = playersObj[uid];

                    const div = document.createElement('div');

                    div.className = 'flex justify-between items-center bg-white p-2 rounded border text-xs';

                    div.innerHTML = `<span>${p.name} ${uid === data.meta.hostUid ? '👑' : ''}</span>

                                     <span class="${p.ready ? 'text-green-600 font-bold' : 'text-gray-400'}">${p.ready ? '已準備 ⭕' : '未準備 ❌'}</span>`;

                    playerListEl.appendChild(div);

                });

                document.getElementById('player-count').innerText = count;

 

                // 遊戲狀態切換檢查

                if (data.meta.status === 'playing' || data.meta.status === 'round_active') {

                    renderGameScreen(data);

                } else if (data.meta.status === 'finished') {

                    renderFinishedScreen(data);

                }

            });

        }

 

        // 切換準備狀態

        function toggleReady() {

            const playerRef = roomRef.child('players/' + currentUserUid + '/ready');

            playerRef.once('value', (snap) => {

                const currentReady = snap.val();

                playerRef.set(!currentReady);

            });

        }

 

        // 主持人開始遊戲

        function startGame() {

            roomRef.child('meta/status').set('playing');

            roomRef.child('game').update({

                currentRound: 1,

                questionIndex: Math.floor(Math.random() * QUESTION_BANK.length)

            });

        }

 

        // 主持人切換下一回合

        function nextRound() {

            roomRef.once('value', (snap) => {

                const data = snap.val();

                const total = data.meta.totalRounds;

                const current = data.game.currentRound;

 

                if (current >= total) {

                    roomRef.child('meta/status').set('finished');

                } else {

                    roomRef.child('game').update({

                        currentRound: current + 1,

                        questionIndex: Math.floor(Math.random() * QUESTION_BANK.length)

                    });

                }

            });

        }

 

        // 渲染遊戲畫面

        function renderGameScreen(data) {

            switchView('view-game');

            const qIndex = data.game.questionIndex || 0;

            const q = QUESTION_BANK[qIndex % QUESTION_BANK.length];

 

            document.getElementById('game-round-indicator').innerText = `回合 ${data.game.currentRound} / ${data.meta.totalRounds}`;

            document.getElementById('question-category').innerText = q.cat;

            document.getElementById('question-title').innerText = q.title;

           

            const contentEl = document.getElementById('question-content');

            const interactionEl = document.getElementById('game-interaction-area');

            interactionEl.innerHTML = '';

 

            if (isHost) {

                document.getElementById('host-game-controls').classList.remove('hidden');

            }

 

            if (q.options) {

                contentEl.innerText = "請在下方選擇您的答案：";

                q.options.forEach(opt => {

                    const btn = document.createElement('button');

                    btn.className = 'w-full bg-partyPurple/10 hover:bg-partyPurple/20 text-partyPurple font-bold py-2.5 px-4 rounded-xl text-sm transition border border-partyPurple/30';

                    btn.innerText = opt;

                    btn.onclick = () => {

                        alert('已送出答案： ' + opt);

                    };

                    interactionEl.appendChild(btn);

                });

            } else {

                contentEl.innerText = "請依照題目說明實際進行挑戰！完成後由主持人判定。";

            }

        }

 

        // 渲染結算畫面

        function renderFinishedScreen(data) {

            switchView('view-finished');

            const resultsEl = document.getElementById('final-results');

            resultsEl.innerHTML = '';

           

            const players = data.players || {};

            Object.values(players).forEach(p => {

                const div = document.createElement('div');

                div.className = 'flex justify-between items-center bg-white p-2 rounded border';

                div.innerHTML = `<span>${p.name}</span><span class="font-bold text-partyPurple">${p.score || 0} 分</span>`;

                resultsEl.appendChild(div);

            });

        }
